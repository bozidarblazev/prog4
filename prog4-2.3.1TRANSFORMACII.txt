#include <iostream>
#include <cmath>
#include <vector>
#include <sstream>
#include <fstream>
using namespace std;
//funkcija za mnozenje na vektor redica so matrica 3x3
void mnozi(const vector<float>& p, const float matrica[3][3], vector<float>& rezultat) {
    // Odi po koloni na matricata
    for (int kol=0;kol<3;kol++) {
           rezultat[kol] = 0;   // Sekogas resetirame pred sobiranje
        //mnozenje na matrici
        for (int red = 0; red < 3; red++) {
            rezultat[kol] += p[red] * matrica[red][kol];
        }
    }
}


//funkcija za rotacija na tocka okolu koordinatniot pocetok
vector<float> rotacija(float x, float y, float agol) {
    vector<float> tochka = {x, y, 1};
    //matrica za rotacija
    float matricaRotacija[3][3] = {
        {cos(agol), -sin(agol), 0},
        {sin(agol),  cos(agol), 0},
        {0, 0, 1}
    };
    vector<float> rezultat(3, 0);
    mnozi(tochka, matricaRotacija, rezultat);

    return rezultat;
}


//funkcija za translacija na dadena tocka
vector<float> translacija(float x, float y, float dx, float dy) {
    vector<float> tochka = {x, y, 1};
    // Matrica za translacija
    float matricaTranslacija[3][3] = {
        {1, 0, 0},
        {0, 1, 0},
        {dx, dy, 1}
    };
    vector<float> rezultat(3, 0);
    mnozi(tochka, matricaTranslacija, rezultat);

    return rezultat;
}


//funkcija za skaliranje
vector<float> skaliranje(float x, float y, float koef) {
    vector<float> tochka = {x, y, 1};
    // Matrica za skaliranje
    float matricaSkaliranje[3][3] = {
        {koef, 0, 0},
        {0, koef, 0},
        {0, 0, 1}
    };
    vector<float> rezultat(3, 0);
    mnozi(tochka, matricaSkaliranje, rezultat);

    return rezultat;
}
